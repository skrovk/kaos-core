"""Fresh process experiments using the production graph host and authority.

Run scripts use existing immutable descriptors through ``run.o``. The runner
owns processes, a transparent network and an evidence log, never K transitions.
Observation timeouts and process teardown cannot establish graph closure.
"""

import argparse
import hashlib
import json
import math
import os
import socket
import subprocess
import threading
import time
from pathlib import Path

import cbor2

from .graph_protocol import Channel, Endpoint, Node, Port, uint
from .graph_relay import GraphRelay
from .graph_transport import CONTENT, CREATED, NOT_FOUND, GraphTransport
from .orchestrator import Orchestrator


def _sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


class GraphRun:
    """One isolated run, up to eight hosts, one fixed authority (99).

    Calls may overlap across threads. Each coordinator has one management
    exchange at a time; no lock is held while waiting for a lifecycle to finish.
    Scripts authorize operations with ``o`` and register immutable artifacts
    with ``add_artifact`` before submission. The transport itself is
    exposed for setup, not concurrent exchanges bypassing this runner's locks.
    """

    authority = 99

    def __init__(self, host, output, devices=(1, 2), *, seed=0, trace_channels=False):
        self.trace_channels = bool(trace_channels)
        self.host = Path(host).resolve()
        self.output = Path(output).resolve()
        self.devices = tuple(uint(device) for device in devices)
        if (
            not 1 <= len(self.devices) <= 8
            or len(set(self.devices)) != len(self.devices)
            or self.authority in self.devices
        ):
            raise ValueError(
                "one through eight distinct devices excluding authority 99"
            )
        self.seed = uint(seed)
        self.o = Orchestrator(self.devices)
        self._locks = {device: threading.Lock() for device in self.devices}
        self._log_lock = threading.Lock()
        self._events = None
        self._relay_overflow_drops = 0
        self._processes = {}
        self._readers = []
        self._reader_errors = []
        self._stderr = {}
        self.network = None
        self.transport = None
        self._entered = False

    def emit(self, kind, **fields):
        with self._log_lock:
            if kind == "packet_dropped" and fields.get("reason") == "relay_overflow":
                self._relay_overflow_drops += 1
            self._events.write(
                json.dumps(
                    {
                        "kind": kind,
                        "timestamp_us": time.monotonic_ns() // 1000,
                        **fields,
                    },
                    separators=(",", ":"),
                )
                + "\n"
            )
            self._events.flush()

    def __enter__(self):
        if self._entered:
            raise RuntimeError("a graph run cannot be restarted")
        self._entered = True
        if not self.host.is_file():
            raise FileNotFoundError(self.host)
        self.output.mkdir(parents=True, exist_ok=False)
        self._events = (self.output / "events.jsonl").open("w")
        reservations = []
        try:
            actual = {}
            for device in self.devices:
                sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
                reservations.append(sock)
                sock.bind(("127.0.0.1", 0))
                actual[device] = sock.getsockname()
            self.network = GraphRelay(
                (*self.devices, self.authority), self.emit, seed=self.seed
            )
            self.transport = GraphTransport(
                ("127.0.0.1", 0),
                {device: self.network.addresses[device] for device in self.devices},
                retry_jitter=lambda: 0,
            )
            actual[self.authority] = self.transport.address
            self.network.start(actual)
            source_dir = Path(__file__).resolve().parent
            sources = sorted(source_dir.glob("graph_*.py")) + [
                source_dir / "orchestrator.py",
                source_dir / "models.py",
            ]
            (self.output / "provenance.json").write_text(
                json.dumps(
                    {
                        "host": {"path": str(self.host), "sha256": _sha(self.host)},
                        "python_sources": {str(path): _sha(path) for path in sources},
                        "devices": self.devices,
                        "authority": self.authority,
                        "seed": self.seed,
                        "actual_addresses": actual,
                        "proxy_addresses": self.network.addresses,
                        "clock": "real monotonic time; host scheduling is not deterministic",
                        "diagnostics": "supplementary stdout; loss count unavailable",
                        "channel_trace": self.trace_channels,
                        "network": "bounded UDP proxy queue; kernel receive loss is not measured",
                    },
                    indent=2,
                )
                + "\n"
            )
            environment = dict(os.environ)
            environment.pop("KAOS_COLLECTOR", None)
            environment.pop("KAOS_TRACE_CHANNELS", None)
            if self.trace_channels:
                environment["KAOS_TRACE_CHANNELS"] = "1"
            environment["ASAN_OPTIONS"] = (
                environment.get("ASAN_OPTIONS", "") + ":detect_stack_use_after_return=0"
            )
            ready = {}
            for device, reservation in zip(self.devices, reservations):
                reservation.close()
                arguments = [
                    str(self.host),
                    str(actual[device][1]),
                    str(self.network.addresses[self.authority][1]),
                    str(device),
                    str(self.authority),
                    *(
                        f"{peer}:{self.network.addresses[peer][1]}"
                        for peer in self.devices
                        if peer != device
                    ),
                ]
                errors = (self.output / f"host-{device}.stderr").open("w")
                self._stderr[device] = errors
                process = subprocess.Popen(
                    arguments,
                    stdout=subprocess.PIPE,
                    stderr=errors,
                    text=True,
                    env=environment,
                )
                self._processes[device] = process
                ready[device] = threading.Event()
                reader = threading.Thread(
                    target=self._read,
                    args=(device, process, actual[device][1], ready[device]),
                    daemon=True,
                )
                self._readers.append(reader)
                reader.start()
            deadline = time.monotonic() + 5
            for device in self.devices:
                if not ready[device].wait(max(0, deadline - time.monotonic())):
                    raise RuntimeError(f"host {device} did not become ready")
            self._check()
            self.emit("run_ready")
            return self
        except BaseException:
            self._close()
            raise
        finally:
            for reservation in reservations:
                reservation.close()

    def _read(self, device, process, port, ready):
        listening = False
        try:
            for line in process.stdout:
                line = line.rstrip("\n")
                self.emit("host_stdout", device=device, line=line)
                if line == f"LISTENING {port}":
                    listening = True
                    ready.set()
            if not listening:
                self._reader_errors.append(
                    RuntimeError(f"host {device} exited before LISTENING")
                )
        except (OSError, ValueError) as error:
            self._reader_errors.append(error)
        finally:
            # EOF before LISTENING lets initialization report the process exit.
            ready.set()

    def _check(self):
        if self._reader_errors:
            raise RuntimeError(
                "host observation reader failed"
            ) from self._reader_errors[0]
        self.network.check()
        for device, process in self._processes.items():
            if process.poll() is not None:
                raise RuntimeError(f"host {device} exited during the experiment")

    def add_artifact(self, reference, data):
        """Register immutable bytes and retain their identity in the run log."""
        data = bytes(data)
        size, checksum = self.transport.add_artifact(reference, data)
        self.emit(
            "artifact",
            reference_hex=bytes(reference).hex(),
            size=size,
            crc32=checksum,
            sha256=hashlib.sha256(data).hexdigest(),
        )
        return size, checksum

    def submit(self, request, *, allowance_us=1_000_000):
        """Submit once. A response or timeout is not an operation outcome."""
        with self._locks[request.coordinator]:
            self._check()
            self.emit(
                "submit",
                operation=request.operation.value,
                coordinator=request.coordinator,
                operation_kind=int(request.kind),
                payload_hex=request.payload.hex(),
            )
            try:
                response = self.transport.submit(
                    self.o, request.operation, allowance_us=allowance_us
                )
            except (TimeoutError, OSError) as error:
                self.emit(
                    "submission_uncertain",
                    operation=request.operation.value,
                    error=str(error),
                )
                raise
            self.emit(
                "admission_response",
                operation=request.operation.value,
                coordinator=request.coordinator,
                code=response.code,
                message_id=response.message_id,
                token_hex=response.token.hex(),
                payload_hex=response.payload.hex(),
            )
            return response

    def query(self, request, *, allowance_us=1_000_000):
        """Retain accepted evidence before sending its exact storage ACK.

        Missing history returns None. Neither it nor transport loss releases
        authorization. An ACK timeout leaves the accepted report available.
        """
        with self._locks[request.coordinator]:
            self._check()
            response = self.transport.query_report(
                request.coordinator, request.operation, allowance_us=allowance_us
            )
            if response.code == NOT_FOUND:
                self.emit("history_missing", operation=request.operation.value)
                return None
            if response.code != CONTENT:
                raise RuntimeError(f"query response code {response.code}")
            acknowledgement = self.o.apply_result(
                response.payload, peer=request.coordinator
            )
            report = cbor2.loads(response.payload)
            self.emit(
                "report",
                operation=request.operation.value,
                coordinator=request.coordinator,
                report=report,
                payload_hex=response.payload.hex(),
            )
            if acknowledgement is not None:
                try:
                    reply = self.transport.acknowledge(
                        request.coordinator, acknowledgement, allowance_us=allowance_us
                    )
                    self.emit(
                        "storage_ack",
                        operation=request.operation.value,
                        revision=report[4],
                        code=reply.code,
                    )
                except (TimeoutError, OSError) as error:
                    self.emit(
                        "storage_ack_uncertain",
                        operation=request.operation.value,
                        error=str(error),
                    )
            return report

    def wait(self, request, *, timeout_s=5):
        """Wait for a non-Pending report; cutoff is only an observation limit."""
        if not math.isfinite(timeout_s) or timeout_s <= 0:
            raise ValueError("observation timeout must be positive and finite")
        deadline = time.monotonic() + timeout_s
        while (remaining := deadline - time.monotonic()) > 0:
            try:
                report = self.query(
                    request, allowance_us=max(1, min(250_000, int(remaining * 500_000)))
                )
            except TimeoutError:
                report = None
            if report is not None and report[5] != 0:
                return report
            time.sleep(min(0.01, max(0, deadline - time.monotonic())))
        self.emit("observation_timeout", operation=request.operation.value)
        raise TimeoutError(f"no observed outcome for {request.operation.value}")

    def _close(self):
        failures = []

        def record(kind, **fields):
            try:
                self.emit(kind, **fields)
            except (OSError, ValueError) as error:
                failures.append(f"observation log failed: {error}")

        for process in self._processes.values():
            if process.poll() is None:
                process.terminate()
        for device, process in self._processes.items():
            try:
                process.wait(timeout=5)
            except subprocess.TimeoutExpired:
                process.kill()
                process.wait(timeout=5)
                failures.append(f"host {device} required forced termination")
            if process.returncode != 0:
                failures.append(f"host {device} exit {process.returncode}")
            record("process_exit", device=device, returncode=process.returncode)
        for reader in self._readers:
            reader.join(timeout=1)
            if reader.is_alive():
                failures.append("host observation reader did not stop")
        for device, stream in self._stderr.items():
            stream.close()
            if (self.output / f"host-{device}.stderr").stat().st_size:
                failures.append(f"host {device} wrote stderr; inspect host log")
        if self.transport is not None:
            self.transport.close()
        if self.network is not None:
            try:
                self.network.close()
                self.network.check()
            except (OSError, RuntimeError) as error:
                failures.append(str(error))
        if self._relay_overflow_drops:
            failures.append(
                f"relay queue overflow dropped {self._relay_overflow_drops} packet copies"
            )
        for process in self._processes.values():
            process.stdout.close()
        if self._reader_errors:
            failures.append("host observation reader failed")
        if self._events is not None:
            record(
                "teardown",
                errors=list(failures),
                proves_graph_cleanup=False,
                relay_overflow_drops=self._relay_overflow_drops,
            )
            self._events.close()
        return failures

    def __exit__(self, exc_type, _exc, _traceback):
        failures = self._close()
        if failures and exc_type is None:
            raise RuntimeError("; ".join(failures))


def run_lifecycle(host, guest, output, *, seed=0):
    """Six operation kinds and explicit fresh binding reuse, with traffic.wasm."""
    outcomes = []
    with GraphRun(host, output, seed=seed) as run:
        artifact = Path(guest).read_bytes()
        size, crc = run.add_artifact(b"traffic", artifact)

        def complete(request):
            if run.submit(request).code != CREATED:
                raise RuntimeError("original request was not admitted")
            report = run.wait(request)
            if report[5] != 1:
                raise RuntimeError(f"operation did not succeed: {report}")
            outcomes.append(
                {
                    "operation": request.operation.value,
                    "kind": int(request.kind),
                    "report": report,
                }
            )
            return request

        source = complete(
            run.o.add_node(
                Node(
                    10,
                    1,
                    b"traffic",
                    size,
                    crc,
                    131072,
                    b"\x00",
                    3_000_000,
                    2_000_000,
                    (Port(1, 1, 3, 4),),
                )
            )
        )
        channel = Channel(
            Endpoint(10, source.instance, 1, 1),
            Endpoint(20, None, 2, 2),
            7,
            3,
            4,
            2,
            3_000_000,
        )
        receiver = complete(
            run.o.add_node(
                Node(
                    20,
                    2,
                    b"traffic",
                    size,
                    crc,
                    131072,
                    b"\x01",
                    3_000_000,
                    2_000_000,
                    (Port(2, 0, 3, 4),),
                ),
                channels=(channel,),
                predicate_tokens=b"\x02\x00",
            )
        )
        complete(run.o.remove_edge(receiver.bindings[0], cleanup_us=2_000_000))
        complete(run.o.add_edge(receiver.bindings[0].channel, cleanup_us=2_000_000))
        complete(run.o.remove_node(receiver.instance, cleanup_us=2_000_000))
        complete(
            run.o.remove_node(source.instance, cleanup_us=2_000_000, no_edges=True)
        )
    result = {
        "status": "passed",
        "outcomes": outcomes,
        "scope": "six graph operations; no ventilation or target qualification",
    }
    (Path(output) / "result.json").write_text(json.dumps(result, indent=2) + "\n")
    return result


def main():
    parser = argparse.ArgumentParser(
        description="Run the six-operation graph workload with actual KaOS hosts"
    )
    parser.add_argument("--host", required=True, type=Path)
    parser.add_argument(
        "--guest", required=True, type=Path, help="traffic.wasm verification fixture"
    )
    parser.add_argument("--output", required=True, type=Path)
    parser.add_argument("--seed", type=int, default=0)
    args = parser.parse_args()
    try:
        result = run_lifecycle(args.host, args.guest, args.output, seed=args.seed)
    except (OSError, RuntimeError, ValueError) as error:
        parser.exit(2, f"graph run failed: {error}\n")
    print(json.dumps(result, indent=2))


if __name__ == "__main__":
    main()
