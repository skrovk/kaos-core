# The same restricted runtime is built as an ESP-IDF component or host library.
option(CORE_WAMR_TRACE_PREPARE "Print qualification arena accounting and preparation diagnostics" OFF)
if(ESP_PLATFORM)
    if(NOT CMAKE_BUILD_EARLY_EXPANSION)
        if(NOT CONFIG_IDF_TARGET_ESP32 OR CONFIG_SPIRAM)
            message(FATAL_ERROR "P3 target profile requires original ESP32 without PSRAM")
        endif()
        if(NOT CONFIG_FREERTOS_USE_TRACE_FACILITY)
            message(FATAL_ERROR "WAMR stack boundary checks require FreeRTOS trace facility")
        endif()
        set(WAMR_BUILD_PLATFORM esp-idf)
        set(WAMR_BUILD_TARGET XTENSA)
        include("${CMAKE_CURRENT_LIST_DIR}/wamr.cmake")
    endif()
    idf_component_register(SRCS "kaos_wamr.c" "profile.c" "native_fixture.c" ${WAMR_RUNTIME_LIB_SOURCE}
        INCLUDE_DIRS "../include" REQUIRES pthread esp_timer lwip)
    target_compile_features(${COMPONENT_LIB} PUBLIC c_std_11)
    if(CORE_FOUR_NODE_PROFILE)
        target_compile_definitions(${COMPONENT_LIB} PUBLIC CORE_FOUR_NODE_PROFILE=1)
    endif()
    if(CORE_WAMR_TRACE_PREPARE)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE CORE_WAMR_TRACE_PREPARE=1)
    endif()
    # Pinned upstream format strings predate the IDF toolchain's int typedefs.
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-format)
    return()
endif()

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "P3 host qualification targets Linux x86_64")
endif()
find_package(Threads REQUIRED)
set(WAMR_BUILD_PLATFORM linux)
set(WAMR_BUILD_TARGET X86_64)
include("${CMAKE_CURRENT_LIST_DIR}/wamr.cmake")
add_library(kaos_wamr_engine STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_link_libraries(kaos_wamr_engine PUBLIC Threads::Threads m dl)
add_library(kaos_wamr STATIC kaos_wamr.c profile.c native_fixture.c)
target_include_directories(kaos_wamr PRIVATE "${WAMR_ROOT_DIR}/core/iwasm/include")
target_include_directories(kaos_wamr PUBLIC "${CMAKE_CURRENT_LIST_DIR}/../include")
target_compile_features(kaos_wamr PUBLIC c_std_11)
if(CORE_FOUR_NODE_PROFILE)
    target_compile_definitions(kaos_wamr PUBLIC CORE_FOUR_NODE_PROFILE=1)
endif()
if(CORE_WAMR_TRACE_PREPARE)
    target_compile_definitions(kaos_wamr PRIVATE CORE_WAMR_TRACE_PREPARE=1)
endif()
set_target_properties(kaos_wamr PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(kaos_wamr PRIVATE -Wall -Wextra -Werror)
target_link_libraries(kaos_wamr PUBLIC kaos_wamr_engine)
target_link_options(kaos_wamr PUBLIC -Wl,-z,noexecstack)
